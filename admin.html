🗄️ SQL úteis
Resumo por status
SELECT status, COUNT(*) FROM psis GROUP BY status;
Psis premium ativas
SELECT nome, crp, email, cidade FROM psis WHERE premium = true AND status = 'aprovado';
Encaminhamentos em aberto
SELECT indicado_por, tipo_buscado, demanda, urgencia, created_at FROM encaminhamentos_cruzados WHERE status = 'disponivel' ORDER BY created_at DESC;
Psis sem e-mail (não conseguem logar no Met)
SELECT nome, crp, handle FROM psis WHERE email IS NULL OR email = '';
